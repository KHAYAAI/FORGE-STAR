# libforge: portable C99 payload core. Standalone build + tests:
#   cmake -S flight/libforge -B build/libforge && cmake --build build/libforge && ctest --test-dir build/libforge
cmake_minimum_required(VERSION 3.16)
project(libforge C)
set(CMAKE_C_STANDARD 99)
add_compile_options(-Wall -Wextra -Werror -pedantic)
add_library(forge STATIC src/forge.c src/forge_auth.c src/forge_orbit.c src/forge_device.c)
target_link_libraries(forge PUBLIC m)
target_include_directories(forge PUBLIC include)
enable_testing()
add_executable(test_libforge test/test_libforge.c)
target_link_libraries(test_libforge forge)
target_include_directories(test_libforge PRIVATE test)
add_test(NAME libforge COMMAND test_libforge)
add_executable(test_auth test/test_auth.c)
target_link_libraries(test_auth forge)
add_test(NAME auth COMMAND test_auth)
add_executable(test_orbit test/test_orbit.c)
target_link_libraries(test_orbit forge)
target_include_directories(test_orbit PRIVATE test)
add_test(NAME orbit COMMAND test_orbit)
add_executable(test_device test/test_device.c)
target_link_libraries(test_device forge)
target_include_directories(test_device PRIVATE test)
add_test(NAME device COMMAND test_device)
